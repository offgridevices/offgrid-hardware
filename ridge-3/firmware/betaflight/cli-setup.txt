# Ridge 3 stack (revision 2) - Betaflight 2025.12 first-time setup
# Paste into the Betaflight CLI after flashing RIDGE3, then check the
# orientation in the Setup tab
# BEFORE the first flight: tilt the nose down and the model must tilt nose
# down; right side down, it must roll right; yaw right and it must yaw right.
# Do not change the board alignment from 0/0/0.
# Most lines below restate the firmware's own defaults, so that a pasted
# setup is complete even after "defaults nosave" or an older dump.

# receiver: CRSF on UART2 (the pads marked 5V G R2 T2), telemetry on
serial UART2 64 115200 57600 0 115200
set serialrx_provider = CRSF
feature TELEMETRY

# motors: AM32 ESC board, bidirectional DShot for the RPM filter.
# The ESC has no TLM line (the lead's pin 4 brings it 3.3 V), so there is
# no ESC serial telemetry
# (leave feature ESC_SENSOR off).  Extended DShot telemetry brings the
# ESCs' temperature, voltage and current back over the motor lines instead.
set motor_pwm_protocol = DSHOT300
set dshot_bidir = ON
set dshot_edt = ON
# XING2 1404 is 9N12P: 12 magnets in the bell.  Set your motor's count.
set motor_poles = 12

# battery voltage: 30k/2k divider on the FC (ratio 16), good for 2S to 6S
set battery_meter = ADC
set vbat_scale = 160
set vbat_min_cell_voltage = 330
set vbat_warning_cell_voltage = 350
set vbat_max_cell_voltage = 430
# battery current: the ESC sends CUR = the average of its four channels'
# 50 mV/A current-sense outputs = 12.5 mV per amp of total battery current,
# 0 A = 0 V.  Betaflight's scale is mV per 10 A, its offset mA.
# Check against a bench supply's ammeter at bring-up; trim ibata_scale if
# needed (a higher scale reads lower current).
set current_meter = ADC
set ibata_scale = 125
set ibata_offset = 0

# blackbox to the 16 MB flash (a board built without it: set blackbox_device = NONE)
set blackbox_device = SPIFLASH

# no beeper on the board: the motors beep instead (DShot beacon), on a lost
# receiver link and on the beeper switch (the two conditions the beacon takes)
set beeper_dshot_beacon_tone = 1
beacon RX_LOST
beacon RX_SET

# ---------------------------------------------------------------- video
# Pick ONE of the two blocks below.  The HD connector and the T1/R1 pads
# are both UART1, so a digital system and an analog VTX's control wire
# cannot be used at the same time.

# --- HD system (DJI O3/O4, Walksnail, HDZero) on the 6-pin HD connector:
# UART1 = "VTX (MSP + DisplayPort)" (131073), the firmware's default.
# vcd_video_system = HD makes Betaflight draw the OSD over MSP DisplayPort.
# Without it, a board with the AT7456E fitted would find the analog chip
# first and send the OSD there, and the goggles would show none.
serial UART1 131073 115200 57600 0 115200
set osd_displayport_device = MSP
set vcd_video_system = HD

# --- Analog camera + VTX instead: comment out the three HD lines above and
# uncomment these.  The OSD is drawn by the AT7456E on the board, between
# the camera and VTX video pads.  AUTO follows the camera; set
# PAL or NTSC to force one.  For SmartAudio (or Tramp: 8192) wire the VTX's
# control line to pad T1.
# serial UART1 2048 115200 57600 0 115200
# set osd_displayport_device = AUTO
# set vcd_video_system = AUTO

# ---------------------------------------------------------------- VTX power
# The 9 V video supply runs from its own battery pads (VBAT and G by the HD
# connector, wired to the ESC's battery pads), never from the stack lead.
# A thermostat on the board switches it off above 96 C at the regulator and
# back on at 76 C, whatever Betaflight does.  It can also be switched off
# from the radio, e.g. to keep the VTX cool on the bench.  PB5 (PINIO1)
# drives an N-FET that pulls the 9 V regulator's enable low:
#   PB5 low  = VTX ON   (the default, and during boot: the FET gate has a
#                        pull-down, so the VTX powers up with the FC)
#   PB5 high = VTX OFF
# PINIO1 follows the USER1 mode (box id 40), not inverted: USER1 active =
# PB5 high = VTX OFF.  These are the firmware's defaults:
set pinio_config = 1,1,1,1
set pinio_box = 40,255,255,255
set box_user_1_name = VTX_OFF
# Then assign USER1 ("VTX_OFF" in the Modes tab) to a switch.  Example,
# VTX off while AUX4 (channel 8) is high; uncomment and adapt:
# aux 5 40 3 1700 2100 0 0
# Do not free PB5 ("resource PINIO 1 NONE"): Betaflight pulls unused pins up,
# which would half-switch the FET.
# On USB power alone the 9 V rail stays off whatever PINIO says: it has no
# battery input then (its own pads).

save
